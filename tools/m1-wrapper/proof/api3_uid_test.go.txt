type uidProofExecutor struct {t *testing.T; dbPath string; uid string; calls int; destroyed bool}
func(e *uidProofExecutor)Name()string{return "harmless-uid-proof"}
func(e *uidProofExecutor)SetChannel(spec.Channel){}
func(e *uidProofExecutor)Exec(u string,ctx context.Context,m *spec.ExpModel)*spec.Response{
 e.calls++
 if u!=e.uid||ctx.Value(spec.Uid)!=e.uid {e.t.Fatal("UID replaced before executor")}
 // An independently opened SQLite connection must already see the UID.
 db,err:=sql.Open("sqlite",e.dbPath);if err!=nil{e.t.Fatal(err)};defer db.Close()
 var saved string;if err=db.QueryRow("SELECT uid FROM experiment WHERE uid=?",u).Scan(&saved);err!=nil||saved!=u{e.t.Fatal("not committed before executor",err)}
 if destroy,ok:=spec.IsDestroy(ctx);ok {if destroy!=u{e.t.Fatal("destroy mismatch")};e.destroyed=true;return spec.ReturnSuccess(true)}
 if m.ActionFlags[UidFlag]!=u{e.t.Fatal("Cobra inherited UID missing")}
 // Current test process exists; no CPU load or external helper is started.
 return spec.ReturnSuccess(os.Getpid())
}
